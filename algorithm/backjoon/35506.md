# Good Bye, 별 찍기!

```python
import sys

input = sys.stdin.readline

def solution():
    N = int(input())
    for i in range(N):
        print(' '*(N*2-i-1)+'*'+' '*N+'*'+' '*(i*2+1)+'*'+' '*(N-i-1))
    for i in range(N):
        print(' '*(N-i-1)+'*'+' '*(N+i*2+1)+'*'+' '*((N-i)*2-1)+'*'+' '*i)

solution()
```
